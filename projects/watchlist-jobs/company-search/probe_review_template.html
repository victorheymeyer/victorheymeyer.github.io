1 — run this against Supabase

Edit the seattle_rec in (...) list to whichever tiers you're reviewing this round.

select ats, slug, company, url, total_jobs, sw_jobs, software_pct,
       local, seattle_rec, seattle_rec_note
from ats_probe_latest
where reviewed is not true
  and seattle_rec in ('Yes-Go', 'Yes-Hold1', 'Yes-Hold2', 'Yes-SmallCoLowSW')
order by
  case seattle_rec
    when 'Yes-Go' then 1
    when 'Yes-Hold1' then 2
    when 'Yes-Hold2' then 3
    when 'Yes-SmallCoLowSW' then 4
  end,
  sw_jobs desc nulls last;
